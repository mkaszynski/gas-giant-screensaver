cmake_minimum_required(VERSION 3.20)
project(gas_giant_screensaver VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GLFW REQUIRED IMPORTED_TARGET glfw3)
pkg_check_modules(GLES REQUIRED IMPORTED_TARGET glesv2)
find_package(PNG REQUIRED)
include(GNUInstallDirs)
add_library(observatory src/system.cpp src/renderer.cpp)
target_include_directories(observatory PUBLIC src)
target_link_libraries(observatory PUBLIC PkgConfig::GLES PNG::PNG)
target_compile_options(observatory PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(observatory PRIVATE OBSERVATORY_DATA_DIR="${CMAKE_INSTALL_FULL_DATADIR}/gas-giant-screensaver")
add_executable(gas-giant-screensaver src/main.cpp)
target_link_libraries(gas-giant-screensaver PRIVATE observatory PkgConfig::GLFW)
install(TARGETS gas-giant-screensaver RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY shaders assets DESTINATION ${CMAKE_INSTALL_DATADIR}/gas-giant-screensaver)
enable_testing()
add_executable(system-tests tests/system_tests.cpp src/system.cpp)
target_include_directories(system-tests PRIVATE src)
add_test(NAME orbital-behavior COMMAND system-tests)
add_test(NAME cli-help COMMAND gas-giant-screensaver --help)
add_test(NAME cli-invalid COMMAND gas-giant-screensaver --fps 0)
set_tests_properties(cli-invalid PROPERTIES WILL_FAIL TRUE)
add_executable(render-tests tests/render_tests.cpp)
target_link_libraries(render-tests PRIVATE observatory PkgConfig::GLFW)
add_executable(ring-render-tests tests/ring_render_tests.cpp)
target_link_libraries(ring-render-tests PRIVATE observatory PkgConfig::GLFW)
# GPU tests are explicitly run under the real compositor or CI's Xvfb/llvmpipe.
add_test(NAME cli-fractional-fps COMMAND gas-giant-screensaver --fps 1.5)
add_test(NAME cli-nonfinite COMMAND gas-giant-screensaver --day-seconds nan)
add_test(NAME cli-missing COMMAND gas-giant-screensaver --width)
add_test(NAME cli-unknown COMMAND gas-giant-screensaver --unknown 1)
set_tests_properties(cli-fractional-fps cli-nonfinite cli-missing cli-unknown PROPERTIES WILL_FAIL TRUE)
add_executable(edge-render-tests tests/edge_render_tests.cpp)
target_link_libraries(edge-render-tests PRIVATE observatory PkgConfig::GLFW)
