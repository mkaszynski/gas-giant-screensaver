cmake_minimum_required(VERSION 3.20)
project(gas_giant_screensaver VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GLFW REQUIRED IMPORTED_TARGET glfw3)
pkg_check_modules(GLES REQUIRED IMPORTED_TARGET glesv2)
find_package(PNG REQUIRED)
include(GNUInstallDirs)
add_library(observatory src/system.cpp src/emissions.cpp src/sky_catalog.cpp src/sky_texture.cpp src/renderer.cpp src/surface.cpp src/surface_renderer.cpp)
target_include_directories(observatory PUBLIC src)
target_link_libraries(observatory PUBLIC PkgConfig::GLES PNG::PNG)
target_compile_options(observatory PRIVATE -Wall -Wextra -Wpedantic)
target_compile_definitions(observatory PRIVATE OBSERVATORY_DATA_DIR="${CMAKE_INSTALL_FULL_DATADIR}/gas-giant-screensaver")
add_executable(gas-giant-screensaver src/main.cpp src/debug_overlay.cpp)
target_link_libraries(gas-giant-screensaver PRIVATE observatory PkgConfig::GLFW)
install(TARGETS gas-giant-screensaver RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY shaders assets DESTINATION ${CMAKE_INSTALL_DATADIR}/gas-giant-screensaver)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/gas-giant-screensaver)
enable_testing()
add_executable(system-tests tests/system_tests.cpp src/system.cpp)
target_include_directories(system-tests PRIVATE src)
add_test(NAME orbital-behavior COMMAND system-tests)
add_test(NAME cli-help COMMAND gas-giant-screensaver --help)
add_test(NAME cli-invalid COMMAND gas-giant-screensaver --fps 0)
set_tests_properties(cli-invalid PROPERTIES WILL_FAIL TRUE)
add_executable(render-tests tests/render_tests.cpp)
target_link_libraries(render-tests PRIVATE observatory PkgConfig::GLFW)
add_executable(ring-render-tests tests/ring_render_tests.cpp)
target_link_libraries(ring-render-tests PRIVATE observatory PkgConfig::GLFW)
# GPU tests are explicitly run under the real compositor or CI's Xvfb/llvmpipe.
add_test(NAME cli-fractional-fps COMMAND gas-giant-screensaver --fps 1.5)
add_test(NAME cli-nonfinite COMMAND gas-giant-screensaver --day-seconds nan)
add_test(NAME cli-missing COMMAND gas-giant-screensaver --width)
add_test(NAME cli-unknown COMMAND gas-giant-screensaver --unknown 1)
set_tests_properties(cli-fractional-fps cli-nonfinite cli-missing cli-unknown PROPERTIES WILL_FAIL TRUE)
add_executable(edge-render-tests tests/edge_render_tests.cpp)
target_link_libraries(edge-render-tests PRIVATE observatory PkgConfig::GLFW)
add_executable(atmosphere-render-tests tests/atmosphere_render_tests.cpp)
target_link_libraries(atmosphere-render-tests PRIVATE observatory PkgConfig::GLFW)
add_executable(scattering-edge-tests tests/scattering_edge_tests.cpp)
target_link_libraries(scattering-edge-tests PRIVATE observatory PkgConfig::GLFW)
add_executable(emission-tests tests/emission_tests.cpp src/emissions.cpp src/system.cpp)
target_include_directories(emission-tests PRIVATE src)
add_test(NAME emission-physics COMMAND emission-tests)
add_test(NAME cli-invalid-weather COMMAND gas-giant-screensaver --weather-time nan)
set_tests_properties(cli-invalid-weather PROPERTIES WILL_FAIL TRUE)
add_executable(emission-render-tests tests/emission_render_tests.cpp)
target_link_libraries(emission-render-tests PRIVATE observatory PkgConfig::GLFW)

foreach(effect IN ITEMS lightning aurora)
  add_test(NAME cli-${effect}-brightness-range COMMAND gas-giant-screensaver --${effect}-brightness 1000001)
  add_test(NAME cli-${effect}-brightness-negative COMMAND gas-giant-screensaver --${effect}-brightness -1)
  add_test(NAME cli-${effect}-brightness-nonfinite COMMAND gas-giant-screensaver --${effect}-brightness nan)
  set_tests_properties(cli-${effect}-brightness-range cli-${effect}-brightness-negative cli-${effect}-brightness-nonfinite PROPERTIES WILL_FAIL TRUE)
  add_test(NAME cli-${effect}-brightness-valid COMMAND gas-giant-screensaver --${effect}-brightness 0 --${effect}-brightness 0.5 --${effect}-brightness 1000000 --help)
endforeach()

add_test(NAME cli-unix-time COMMAND gas-giant-screensaver --time 1790900000 --weather-time 1790900000 --help)

add_executable(sky-catalog-tests tests/sky_catalog_tests.cpp src/sky_catalog.cpp)
target_include_directories(sky-catalog-tests PRIVATE src)
target_compile_definitions(sky-catalog-tests PRIVATE SKY_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets/sky")
add_test(NAME sky-catalog COMMAND sky-catalog-tests)
add_executable(sky-texture-tests tests/sky_texture_tests.cpp src/sky_texture.cpp)
target_include_directories(sky-texture-tests PRIVATE src)
target_compile_definitions(sky-texture-tests PRIVATE SKY_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets/sky")
add_test(NAME sky-texture COMMAND sky-texture-tests)
add_test(NAME cli-sky-options COMMAND gas-giant-screensaver --no-galaxy --no-stars --uncompressed-galaxy --help)
add_executable(sky-render-tests tests/sky_render_tests.cpp)
target_link_libraries(sky-render-tests PRIVATE observatory PkgConfig::GLFW)

# Optional development benchmark. Production builds contain no profiling calls.
option(OBSERVATORY_BUILD_SKY_BENCHMARK "Build the isolated sky GPU benchmark" OFF)
if(OBSERVATORY_BUILD_SKY_BENCHMARK)
  add_executable(sky-benchmark tests/sky_benchmark.cpp src/renderer.cpp src/sky_catalog.cpp src/sky_texture.cpp src/system.cpp src/emissions.cpp src/surface.cpp src/surface_renderer.cpp)
  target_include_directories(sky-benchmark PRIVATE src)
  target_link_libraries(sky-benchmark PRIVATE PkgConfig::GLFW PkgConfig::GLES PNG::PNG)
  target_compile_definitions(sky-benchmark PRIVATE OBSERVATORY_SKY_PROFILE OBSERVATORY_DATA_DIR="${CMAKE_INSTALL_FULL_DATADIR}/gas-giant-screensaver")
endif()

add_executable(debug-timeline-tests tests/debug_timeline_tests.cpp)
target_include_directories(debug-timeline-tests PRIVATE src)

  set_tests_properties(cli-debug-${mode} PROPERTIES WILL_FAIL TRUE)
endforeach()

target_include_directories(surface-tests PRIVATE src)

  foreach(value IN ITEMS nan -1 10000000001)
    add_test(NAME cli-${option}-${value} COMMAND gas-giant-screensaver --${option} ${value})
    set_tests_properties(cli-${option}-${value} PROPERTIES WILL_FAIL TRUE)
  endforeach()

add_test(NAME cli-travel-valid COMMAND gas-giant-screensaver --travel-time 259200 --travel-speed 0 --fixed-view --help)
add_executable(surface-render-tests tests/surface_render_tests.cpp)
target_link_libraries(surface-render-tests PRIVATE observatory PkgConfig::GLFW)
add_executable(surface-shadow-tests tests/surface_shadow_tests.cpp)
target_link_libraries(surface-shadow-tests PRIVATE observatory PkgConfig::GLFW)

if(OBSERVATORY_BUILD_FLIGHT_BAKER)
  add_executable(flight-baker scripts/bake_flight.cpp src/surface.cpp src/system.cpp)
  target_include_directories(flight-baker PRIVATE src)
endif()
